  380 |    * Resilient POST with Promise.race timeout + exponential backoff for 500/503.
  381 |    * Gracefully swallows "target closed" errors so a crashed page doesn't kill the suite.
  382 |    */
  383 |   async safePost(url: string, options: { data: any, headers: any, label: string }, timeoutMs = 30000): Promise<any> {
  384 |     let lastError: any = null;
  385 |     const startTime = performance.now();
  386 | 
  387 |     for (let attempt = 1; attempt <= 3; attempt++) {
  388 |       try {
  389 |         const response = await this.withTimeout(
  390 |           this.page.request.post(url, { data: options.data, headers: options.headers }),
  391 |           timeoutMs,
  392 |           options.label
  393 |         );
  394 | 
  395 |         if (response.ok()) return response;
  396 | 
  397 |         const status = response.status();
  398 |         const text = await response.text();
  399 |         lastError = { status, text };
  400 | 
  401 |         // Exponential backoff only for transient server errors
  402 |         if (status === 500 || status === 503) {
  403 |           const backoff = attempt * attempt * 1000; // 1s, 4s, 9s
  404 |           Logger.warn(`${options.label} → ${status}. Retry ${attempt}/3 in ${backoff}ms...`);
  405 |           await this.page.waitForTimeout(backoff);
  406 |           continue;
  407 |         }
  408 | 
  409 |         // Non-retryable error (e.g. 4xx) — attach to Allure and return
  410 |         const duration = performance.now() - startTime;
  411 |         await this.attachApiFailureToAllure('POST', url, options.headers, options.data, status, text, duration);
  412 |         return response; // 4xx — return as-is, no retry
  413 | 
  414 |       } catch (err: any) {
  415 |         // Gracefully handle page/context closed — don't crash the suite
  416 |         if (
  417 |           err.message?.includes('Target page, context or browser has been closed') ||
  418 |           err.message?.includes('page has been closed') ||
  419 |           err.message?.includes('context was destroyed')
  420 |         ) {
  421 |           Logger.warn(`${options.label} → Page closed mid-request (attempt ${attempt}). Skipping.`);
  422 |           return { ok: () => false, status: () => 0, text: async () => 'page-closed', json: async () => ({}) };
  423 |         }
  424 | 
  425 |         // Timeout hit — no point retrying a deadlocked backend immediately
  426 |         if (err.message?.includes('[TIMEOUT]')) {
  427 |           Logger.warn(err.message);
  428 |           lastError = { status: 408, text: err.message };
  429 |           break;
  430 |         }
  431 | 
  432 |         lastError = { status: 0, text: err.message };
  433 |         await this.page.waitForTimeout(attempt * 1000);
  434 |       }
  435 |     }
  436 | 
  437 |     // All retries failed
  438 |     const duration = performance.now() - startTime;
  439 |     await this.attachApiFailureToAllure('POST', url, options.headers, options.data, lastError?.status ?? 0, lastError?.text ?? '', duration);
  440 | 
  441 |     return {
  442 |       ok: () => false,
  443 |       status: () => lastError?.status ?? 0,
  444 |       text: async () => lastError?.text ?? '',
  445 |       json: async () => { try { return JSON.parse(lastError?.text ?? '{}'); } catch { return {}; } }
  446 |     };
  447 |   }
